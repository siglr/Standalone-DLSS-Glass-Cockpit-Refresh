  set(defs -D DISPLAY_SLOT=${slot})

  if(kind MATCHES "vr")
   list(APPEND defs -D VR=1)
   set(name map_ps_vr_${slot})
  endif()
  if(kind STREQUAL "t0")
   set(name map_ps_t0_${slot})
  elseif(kind STREQUAL "vr_t0")
   set(name map_ps_vr_t0_${slot})
  endif()
  if(kind MATCHES "t0")
   list(APPEND defs -D SCREEN_T0=1)
  endif()
  add_custom_command(OUTPUT "${GEN}/${name}.h" COMMAND "${DXC}" -T ps_6_6 -HV 2021 -E main ${defs} -WX -O3 -Fh "${GEN}/${name}.h" -Vn ${name} "${G}/src/display_map.hlsl" DEPENDS "${G}/src/display_map.hlsl" VERBATIM)

 endforeach()
endforeach()

# Always stage current DLLs, including incremental/target-only builds where
# the fixture executable itself does not need relinking.
add_custom_target(stage_fixture_runtime
 COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:UnifiedVRGraphics>" "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:UnifiedVRBootstrap>" "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>"
 COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>/OptiScaler"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:fake_ffx>" "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>/OptiScaler/amd_fidelityfx_upscaler_dx12.dll"
 COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>/NVIDIA/NGX/cache"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:fake_proxy>" "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>/nvngx_feature.dll"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:fake_proxy>" "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>/downloaded_model.bin"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:fake_proxy>" "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>/NVIDIA/NGX/cache/feature.dll"
 DEPENDS UnifiedVRGraphics UnifiedVRBootstrap fake_ffx fake_proxy fake_nvidia_feature)
